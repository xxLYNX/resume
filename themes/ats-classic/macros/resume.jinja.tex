{# Visible http(s) label: scheme stripped; full URL stays in \href. #}
{% macro url_display(u) -%}
{%- if u -%}
{%- if u is startingwith("https://") -%}
{%- set v = u[8:] -%}
{%- elif u is startingwith("http://") -%}
{%- set v = u[7:] -%}
{%- else -%}
{%- set v = u -%}
{%- endif -%}
{%- if v is endingwith("/") -%}
{{- v[:-1] -}}
{%- else -%}
{{- v -}}
{%- endif -%}
{%- endif -%}
{%- endmacro %}

{% macro project_block(row) -%}
      \resumeProjectHeading
          {%- if row.resume_tags -%}
          {\textbf{ {{- row.name | latex_escape -}} } $|$ \emph{ {{- row.resume_tags | latex_escape -}} }}{ {{- row.date_range | format_date_range -}} }
          {%- else -%}
          {\textbf{ {{- row.name | latex_escape -}} }}{ {{- row.date_range | format_date_range -}} }
          {%- endif -%}
{%- if row.details %}
          \resumeItemListStart
{%- for line in row.details %}
            \resumeItem{ {{- line | latex_escape -}} }
{%- endfor %}
          \resumeItemListEnd
{%- endif %}
{%- endmacro %}
